## 示例1: 开发一个统计卡片组件

### 需求描述

创建一个用于Dashboard的统计卡片组件，显示标题、数值、增长率和图标。

### 组件设计

**位置**: `packages/effects/common-ui/src/components/statistic-card/`

**功能特性**:

- 显示统计数值和标题
- 显示增长趋势(上升/下降)
- 支持自定义图标和颜色
- 响应式设计

### 实现步骤

#### 1. 创建目录和文件

```bash
mkdir -p packages/effects/common-ui/src/components/statistic-card
cd packages/effects/common-ui/src/components/statistic-card
touch statistic-card.vue types.ts index.ts
```

#### 2. 定义类型 (types.ts)

```typescript
export interface StatisticCardProps {
  /** 标题 */
  title: string;
  /** 数值 */
  value: number | string;
  /** 前缀 */
  prefix?: string;
  /** 后缀 */
  suffix?: string;
  /** 增长率 */
  growth?: number;
  /** 图标 */
  icon?: string;
  /** 主题色 */
  color?: 'primary' | 'success' | 'warning' | 'error';
  /** 是否加载中 */
  loading?: boolean;
}

export type TrendType = 'up' | 'down' | 'neutral';
```

#### 3. 实现组件 (statistic-card.vue)

```vue
<script setup lang="ts">
import type { StatisticCardProps, TrendType } from './types';
import { computed } from 'vue';
import { cn } from '@vben-core/shared/utils';
import { VbenIcon, VbenCard, VbenSpin } from '@vben-core/shadcn-ui';
import { ArrowUp, ArrowDown, Minus } from '@vben-core/icons';

defineOptions({
  name: 'StatisticCard',
});

const props = withDefaults(defineProps<StatisticCardProps>(), {
  color: 'primary',
  loading: false,
});

// 计算趋势类型
const trend = computed<TrendType>(() => {
  if (!props.growth) return 'neutral';
  return props.growth > 0 ? 'up' : 'down';
});

// 趋势图标
const TrendIcon = computed(() => {
  switch (trend.value) {
    case 'up':
      return ArrowUp;
    case 'down':
      return ArrowDown;
    default:
      return Minus;
  }
});

// 趋势颜色
const trendColor = computed(() => {
  switch (trend.value) {
    case 'up':
      return 'text-green-500';
    case 'down':
      return 'text-red-500';
    default:
      return 'text-gray-500';
  }
});

// 主题色映射
const colorClasses = computed(() => {
  const colors = {
    primary: 'bg-blue-50 text-blue-600 dark:bg-blue-950 dark:text-blue-400',
    success: 'bg-green-50 text-green-600 dark:bg-green-950 dark:text-green-400',
    warning:
      'bg-orange-50 text-orange-600 dark:bg-orange-950 dark:text-orange-400',
    error: 'bg-red-50 text-red-600 dark:bg-red-950 dark:text-red-400',
  };
  return colors[props.color];
});
</script>

<template>
  <VbenCard :class="cn('relative overflow-hidden', $attrs.class)">
    <VbenSpin :spinning="loading">
      <div class="flex items-center justify-between">
        <!-- 左侧内容 -->
        <div class="flex-1">
          <p class="text-muted-foreground mb-2 text-sm">{{ title }}</p>

          <div class="mb-2 flex items-baseline gap-1">
            <span v-if="prefix" class="text-lg font-medium">{{ prefix }}</span>
            <span class="text-3xl font-bold">{{ value }}</span>
            <span v-if="suffix" class="text-lg font-medium">{{ suffix }}</span>
          </div>

          <!-- 增长率 -->
          <div v-if="growth !== undefined" class="flex items-center gap-1">
            <component :is="TrendIcon" :class="cn('h-4 w-4', trendColor)" />
            <span :class="cn('text-sm font-medium', trendColor)">
              {{ Math.abs(growth) }}%
            </span>
            <span class="text-muted-foreground text-xs">vs last period</span>
          </div>
        </div>

        <!-- 右侧图标 -->
        <div
          v-if="icon"
          :class="
            cn(
              'flex h-14 w-14 items-center justify-center rounded-full',
              colorClasses,
            )
          "
        >
          <VbenIcon :icon="icon" class="text-2xl" />
        </div>
      </div>

      <!-- 装饰性背景 -->
      <div
        class="absolute -right-4 -top-4 h-24 w-24 rounded-full opacity-5"
        :class="`bg-${color}-500`"
      />
    </VbenSpin>
  </VbenCard>
</template>
```

## 示例2: 开发一个树形选择器

### 需求描述

创建一个支持异步加载的树形选择器，用于部门/组织选择。

### 组件设计

**位置**: `packages/effects/common-ui/src/components/dept-tree-select/`

### 实现步骤

#### 1. 定义类型 (types.ts)

```typescript
export interface TreeNode {
  id: string;
  name: string;
  parentId?: string;
  children?: TreeNode[];
  disabled?: boolean;
  isLeaf?: boolean;
}

export interface DeptTreeSelectProps {
  /** v-model绑定值 */
  modelValue?: string | string[];
  /** 占位符 */
  placeholder?: string;
  /** 是否多选 */
  multiple?: boolean;
  /** 是否禁用 */
  disabled?: boolean;
  /** 异步加载函数 */
  loadData?: (node: TreeNode) => Promise<TreeNode[]>;
  /** 初始数据 */
  initialData?: TreeNode[];
  /** 是否显示搜索框 */
  showSearch?: boolean;
  /** 是否可清除 */
  clearable?: boolean;
}

export interface DeptTreeSelectEmits {
  'update:modelValue': [value: string | string[]];
  change: [value: string | string[], nodes: TreeNode[]];
}
```

#### 2. 实现组件 (dept-tree-select.vue)

```vue
<script setup lang="ts">
import type {
  DeptTreeSelectProps,
  DeptTreeSelectEmits,
  TreeNode,
} from './types';
import { computed, ref, watch } from 'vue';
import { VbenTree } from '@vben/common-ui';
import { VbenSelect, VbenInput } from '@vben-core/shadcn-ui';

defineOptions({
  name: 'DeptTreeSelect',
});

const props = withDefaults(defineProps<DeptTreeSelectProps>(), {
  placeholder: '请选择部门',
  multiple: false,
  disabled: false,
  showSearch: true,
  clearable: true,
});

const emit = defineEmits<DeptTreeSelectEmits>();

const treeData = ref<TreeNode[]>(props.initialData || []);
const selectedKeys = ref<string[]>([]);
const searchValue = ref('');
const loading = ref(false);

// 同步 v-model
watch(
  () => props.modelValue,
  (val) => {
    if (Array.isArray(val)) {
      selectedKeys.value = val;
    } else if (val) {
      selectedKeys.value = [val];
    } else {
      selectedKeys.value = [];
    }
  },
  { immediate: true },
);

// 过滤树节点
const filteredTreeData = computed(() => {
  if (!searchValue.value) return treeData.value;

  const filterNodes = (nodes: TreeNode[]): TreeNode[] => {
    return nodes.reduce((acc, node) => {
      const match = node.name
        .toLowerCase()
        .includes(searchValue.value.toLowerCase());
      const filteredChildren = node.children ? filterNodes(node.children) : [];

      if (match || filteredChildren.length > 0) {
        acc.push({
          ...node,
          children:
            filteredChildren.length > 0 ? filteredChildren : node.children,
        });
      }
      return acc;
    }, [] as TreeNode[]);
  };

  return filterNodes(treeData.value);
});

// 异步加载子节点
const handleLoadData = async (node: TreeNode) => {
  if (!props.loadData) return;

  loading.value = true;
  try {
    const children = await props.loadData(node);
    // 更新节点的children
    const updateNode = (nodes: TreeNode[]): TreeNode[] => {
      return nodes.map((n) => {
        if (n.id === node.id) {
          return { ...n, children };
        }
        if (n.children) {
          return { ...n, children: updateNode(n.children) };
        }
        return n;
      });
    };

    treeData.value = updateNode(treeData.value);
  } finally {
    loading.value = false;
  }
};

// 选择节点
const handleSelect = (keys: string[], nodes: TreeNode[]) => {
  selectedKeys.value = keys;

  const value = props.multiple ? keys : keys[0] || '';
  emit('update:modelValue', value);
  emit('change', value, nodes);
};

// 获取选中节点
const getSelectedNodes = (nodes: TreeNode[], keys: string[]): TreeNode[] => {
  const result: TreeNode[] = [];

  const findNodes = (items: TreeNode[]) => {
    items.forEach((node) => {
      if (keys.includes(node.id)) {
        result.push(node);
      }
      if (node.children) {
        findNodes(node.children);
      }
    });
  };

  findNodes(nodes);
  return result;
};

// 显示的标签
const selectedLabels = computed(() => {
  const nodes = getSelectedNodes(treeData.value, selectedKeys.value);
  return nodes.map((n) => n.name).join(', ');
});

// 清除选择
const handleClear = () => {
  selectedKeys.value = [];
  emit('update:modelValue', props.multiple ? [] : '');
  emit('change', props.multiple ? [] : '', []);
};
</script>

<template>
  <div class="relative">
    <VbenInput
      :model-value="selectedLabels"
      :placeholder="placeholder"
      :disabled="disabled"
      readonly
      class="cursor-pointer"
    >
      <template #suffix>
        <span
          v-if="clearable && selectedLabels"
          class="text-muted-foreground hover:text-foreground cursor-pointer"
          @click.stop="handleClear"
        >
          ✕
        </span>
        <span v-else class="text-muted-foreground">▼</span>
      </template>
    </VbenInput>

    <!-- 下拉树 -->
    <Teleport to="body">
      <div
        v-show="/* 显示逻辑 */ false"
        class="bg-popover text-popover-foreground absolute z-50 rounded-md border shadow-md"
      >
        <VbenInput
          v-if="showSearch"
          v-model="searchValue"
          placeholder="搜索..."
          class="border-b"
        />

        <VbenTree
          :data="filteredTreeData"
          :selected-keys="selectedKeys"
          :multiple="multiple"
          :load-data="handleLoadData"
          :loading="loading"
          @select="handleSelect"
        />
      </div>
    </Teleport>
  </div>
</template>
```

## 示例3: 开发一个文件上传组件

### 需求描述

创建一个功能完整的文件上传组件，支持拖拽、预览、进度显示。

### 组件设计

**位置**: `packages/effects/common-ui/src/components/file-upload/`

### 实现步骤

#### 1. 定义类型 (types.ts)

```typescript
export interface UploadFile {
  uid: string;
  name: string;
  size: number;
  type: string;
  url?: string;
  status: 'uploading' | 'success' | 'error' | 'pending';
  progress: number;
  response?: any;
  error?: string;
}

export interface FileUploadProps {
  /** v-model绑定值 */
  modelValue?: UploadFile[];
  /** 上传地址 */
  action: string;
  /** 请求头 */
  headers?: Record<string, string>;
  /** 额外参数 */
  data?: Record<string, any>;
  /** 接受的文件类型 */
  accept?: string;
  /** 最大文件大小(MB) */
  maxSize?: number;
  /** 最大文件数量 */
  maxCount?: number;
  /** 是否支持多选 */
  multiple?: boolean;
  /** 是否禁用 */
  disabled?: boolean;
  /** 是否显示列表 */
  showList?: boolean;
  /** 列表类型 */
  listType?: 'text' | 'picture' | 'picture-card';
  /** 自定义上传 */
  customRequest?: (options: UploadRequestOptions) => Promise<any>;
}

export interface UploadRequestOptions {
  file: File;
  action: string;
  headers?: Record<string, string>;
  data?: Record<string, any>;
  onProgress?: (percent: number) => void;
  onSuccess?: (response: any) => void;
  onError?: (error: Error) => void;
}

export interface FileUploadEmits {
  'update:modelValue': [files: UploadFile[]];
  change: [files: UploadFile[]];
  success: [file: UploadFile, response: any];
  error: [file: UploadFile, error: Error];
  preview: [file: UploadFile];
  remove: [file: UploadFile];
}
```

#### 2. 实现上传逻辑 (use-upload.ts)

```typescript
import { ref } from 'vue';
import type { UploadFile, UploadRequestOptions } from './types';
import axios from 'axios';

export function useUpload() {
  const fileList = ref<UploadFile[]>([]);

  // 默认上传请求
  const defaultUpload = async (options: UploadRequestOptions) => {
    const { file, action, headers, data, onProgress, onSuccess, onError } =
      options;

    const formData = new FormData();
    formData.append('file', file);

    if (data) {
      Object.keys(data).forEach((key) => {
        formData.append(key, data[key]);
      });
    }

    try {
      const response = await axios.post(action, formData, {
        headers: {
          'Content-Type': 'multipart/form-data',
          ...headers,
        },
        onUploadProgress: (progressEvent) => {
          if (progressEvent.total) {
            const percent = Math.round(
              (progressEvent.loaded * 100) / progressEvent.total,
            );
            onProgress?.(percent);
          }
        },
      });

      onSuccess?.(response.data);
      return response.data;
    } catch (error) {
      onError?.(error as Error);
      throw error;
    }
  };

  // 添加文件到列表
  const addFile = (file: File): UploadFile => {
    const uploadFile: UploadFile = {
      uid: `${Date.now()}-${Math.random()}`,
      name: file.name,
      size: file.size,
      type: file.type,
      status: 'pending',
      progress: 0,
    };

    fileList.value.push(uploadFile);
    return uploadFile;
  };

  // 更新文件状态
  const updateFile = (uid: string, updates: Partial<UploadFile>) => {
    const index = fileList.value.findIndex((f) => f.uid === uid);
    if (index > -1) {
      fileList.value[index] = { ...fileList.value[index], ...updates };
    }
  };

  // 移除文件
  const removeFile = (uid: string) => {
    const index = fileList.value.findIndex((f) => f.uid === uid);
    if (index > -1) {
      fileList.value.splice(index, 1);
    }
  };

  return {
    fileList,
    defaultUpload,
    addFile,
    updateFile,
    removeFile,
  };
}
```

#### 3. 实现组件 (file-upload.vue)

```vue
<script setup lang="ts">
import type { FileUploadProps, FileUploadEmits, UploadFile } from './types';
import { ref, computed } from 'vue';
import { useUpload } from './use-upload';
import { VbenButton, VbenProgress, VbenIcon } from '@vben-core/shadcn-ui';
import { Upload, X, Eye, FileText } from '@vben-core/icons';

defineOptions({
  name: 'FileUpload',
});

const props = withDefaults(defineProps<FileUploadProps>(), {
  maxSize: 10,
  maxCount: 5,
  multiple: true,
  disabled: false,
  showList: true,
  listType: 'text',
});

const emit = defineEmits<FileUploadEmits>();

const { fileList, defaultUpload, addFile, updateFile, removeFile } =
  useUpload();

const inputRef = ref<HTMLInputElement>();
const isDragging = ref(false);

// 是否达到最大数量
const isMaxCount = computed(() => {
  return fileList.value.length >= props.maxCount;
});

// 触发文件选择
const handleClick = () => {
  if (props.disabled || isMaxCount.value) return;
  inputRef.value?.click();
};

// 文件选择
const handleFileChange = (e: Event) => {
  const input = e.target as HTMLInputElement;
  const files = Array.from(input.files || []);
  handleFiles(files);
  input.value = ''; // 清空以支持重复选择同一文件
};

// 拖拽进入
const handleDragEnter = (e: DragEvent) => {
  e.preventDefault();
  if (!props.disabled) {
    isDragging.value = true;
  }
};

// 拖拽离开
const handleDragLeave = (e: DragEvent) => {
  e.preventDefault();
  isDragging.value = false;
};

// 拖拽放置
const handleDrop = (e: DragEvent) => {
  e.preventDefault();
  isDragging.value = false;

  if (props.disabled) return;

  const files = Array.from(e.dataTransfer?.files || []);
  handleFiles(files);
};

// 处理文件
const handleFiles = async (files: File[]) => {
  // 过滤文件
  const validFiles = files.filter((file) => {
    // 检查文件大小
    if (file.size > props.maxSize * 1024 * 1024) {
      console.error(`文件 ${file.name} 超过最大大小 ${props.maxSize}MB`);
      return false;
    }

    // 检查文件类型
    if (props.accept) {
      const acceptTypes = props.accept.split(',').map((t) => t.trim());
      const isAccepted = acceptTypes.some((type) => {
        if (type.startsWith('.')) {
          return file.name.endsWith(type);
        }
        return file.type.match(type.replace('*', '.*'));
      });

      if (!isAccepted) {
        console.error(`文件 ${file.name} 类型不匹配`);
        return false;
      }
    }

    return true;
  });

  // 限制数量
  const allowCount = props.maxCount - fileList.value.length;
  const filesToUpload = validFiles.slice(0, allowCount);

  // 上传文件
  for (const file of filesToUpload) {
    await uploadFile(file);
  }
};

// 上传单个文件
const uploadFile = async (file: File) => {
  const uploadFile = addFile(file);

  updateFile(uploadFile.uid, { status: 'uploading' });

  const uploadFn = props.customRequest || defaultUpload;

  try {
    const response = await uploadFn({
      file,
      action: props.action,
      headers: props.headers,
      data: props.data,
      onProgress: (percent) => {
        updateFile(uploadFile.uid, { progress: percent });
      },
      onSuccess: (res) => {
        updateFile(uploadFile.uid, {
          status: 'success',
          progress: 100,
          response: res,
          url: res.url,
        });
        emit('success', uploadFile, res);
      },
      onError: (error) => {
        updateFile(uploadFile.uid, {
          status: 'error',
          error: error.message,
        });
        emit('error', uploadFile, error);
      },
    });

    emit('update:modelValue', fileList.value);
    emit('change', fileList.value);
  } catch (error) {
    console.error('上传失败:', error);
  }
};

// 移除文件
const handleRemove = (file: UploadFile) => {
  removeFile(file.uid);
  emit('update:modelValue', fileList.value);
  emit('change', fileList.value);
  emit('remove', file);
};

// 预览文件
const handlePreview = (file: UploadFile) => {
  emit('preview', file);
};

// 获取文件图标
const getFileIcon = (file: UploadFile) => {
  if (file.type.startsWith('image/')) {
    return file.url;
  }
  return FileText;
};
</script>

<template>
  <div class="file-upload">
    <!-- 上传区域 -->
    <div
      v-if="!isMaxCount"
      class="border-border hover:border-primary flex cursor-pointer flex-col items-center justify-center rounded-lg border-2 border-dashed p-8 transition-colors"
      :class="{
        'border-primary bg-primary/5': isDragging,
        'cursor-not-allowed opacity-50': disabled,
      }"
      @click="handleClick"
      @dragenter="handleDragEnter"
      @dragover.prevent
      @dragleave="handleDragLeave"
      @drop="handleDrop"
    >
      <Upload class="text-muted-foreground mb-2 h-12 w-12" />
      <p class="text-muted-foreground mb-1 text-sm">点击上传或拖拽文件到此处</p>
      <p class="text-muted-foreground text-xs">
        支持 {{ accept || '所有类型' }}，最大 {{ maxSize }}MB，最多
        {{ maxCount }} 个文件
      </p>
    </div>

    <!-- 隐藏的文件输入 -->
    <input
      ref="inputRef"
      type="file"
      :accept="accept"
      :multiple="multiple"
      :disabled="disabled"
      class="hidden"
      @change="handleFileChange"
    />

    <!-- 文件列表 -->
    <div v-if="showList && fileList.length > 0" class="mt-4 space-y-2">
      <div
        v-for="file in fileList"
        :key="file.uid"
        class="bg-muted flex items-center gap-3 rounded-lg p-3"
      >
        <!-- 文件图标/缩略图 -->
        <div class="flex-shrink-0">
          <img
            v-if="listType === 'picture' && file.url"
            :src="file.url"
            class="h-10 w-10 rounded object-cover"
          />
          <component
            v-else
            :is="getFileIcon(file)"
            class="text-muted-foreground h-10 w-10"
          />
        </div>

        <!-- 文件信息 -->
        <div class="min-w-0 flex-1">
          <p class="truncate text-sm font-medium">{{ file.name }}</p>
          <p class="text-muted-foreground text-xs">
            {{ (file.size / 1024 / 1024).toFixed(2) }} MB
          </p>

          <!-- 进度条 -->
          <VbenProgress
            v-if="file.status === 'uploading'"
            :value="file.progress"
            class="mt-1"
          />

          <!-- 错误信息 -->
          <p v-if="file.status === 'error'" class="mt-1 text-xs text-red-500">
            {{ file.error }}
          </p>
        </div>

        <!-- 操作按钮 -->
        <div class="flex gap-2">
          <VbenButton
            v-if="file.status === 'success' && file.url"
            size="sm"
            variant="ghost"
            @click="handlePreview(file)"
          >
            <Eye class="h-4 w-4" />
          </VbenButton>

          <VbenButton size="sm" variant="ghost" @click="handleRemove(file)">
            <X class="h-4 w-4" />
          </VbenButton>
        </div>
      </div>
    </div>
  </div>
</template>
```

#### 2. 创建ECharts组件

**位置**: `packages/effects/plugins/src/echarts/`

```typescript
// use-echarts.ts
import { ref, onMounted, onUnmounted, watch, unref, nextTick } from 'vue';
import type { Ref } from 'vue';
import * as echarts from 'echarts/core';
import type { EChartsOption } from 'echarts';

export function useECharts(
  elRef: Ref<HTMLDivElement | null>,
  options?: {
    theme?: 'light' | 'dark';
  },
) {
  const chartInstance = ref<echarts.ECharts>();
  const resizeObserver = ref<ResizeObserver>();

  // 初始化图表
  const initChart = async () => {
    await nextTick();
    const el = unref(elRef);
    if (!el) return;

    chartInstance.value = echarts.init(el, options?.theme);

    // 监听容器大小变化
    resizeObserver.value = new ResizeObserver(() => {
      chartInstance.value?.resize();
    });
    resizeObserver.value.observe(el);
  };

  // 设置图表配置
  const setOption = (option: EChartsOption) => {
    chartInstance.value?.setOption(option);
  };

  // 销毁图表
  const dispose = () => {
    chartInstance.value?.dispose();
    resizeObserver.value?.disconnect();
  };

  onMounted(() => {
    initChart();
  });

  onUnmounted(() => {
    dispose();
  });

  return {
    chartInstance,
    setOption,
    dispose,
  };
}
```

```vue
<!-- chart.vue -->
<script setup lang="ts">

import { ref, watch } from 'vue';
import { useECharts } from './use-echarts';

interface Props {
  option: EChartsOption;
  theme?: 'light' | 'dark';
  height?: string;
  loading?: boolean;
}

const props = withDefaults(defineProps<Props>(), {
  theme: 'light',
  height: '400px',
  loading: false,
});

const chartRef = ref<HTMLDivElement>();
const { setOption, chartInstance } = useECharts(chartRef, {
  theme: props.theme,
});

// 监听配置变化
watch(
  () => props.option,
  (option) => {
    if (option) {
      setOption(option);
    }
  },
  { deep: true, immediate: true },
);

// 监听加载状态
watch(
  () => props.loading,
  (loading) => {
    if (loading) {
      chartInstance.value?.showLoading();
    } else {
      chartInstance.value?.hideLoading();
    }
  },
);
</script>

<template>
  <div ref="chartRef" :style="{ height }" />
</template>
```

## 示例5: 开发一个表格查询组件

### 需求描述

创建一个集成查询表单和数据表格的组件，常用于列表页面。

### 组件设计

**位置**: `packages/effects/common-ui/src/components/table-pro/`

### 实现代码

```vue
<script setup lang="ts" generic="T extends Record<string, any>">
import type { VbenFormSchema } from '@vben-core/form-ui';
import { ref, watch } from 'vue';
import { useVbenForm } from '@vben-core/form-ui';
import { VbenButton, VbenTable } from '@vben-core/shadcn-ui';
import { Page } from '@vben/common-ui';

interface TableProProps<T> {
  /** 查询表单配置 */
  searchSchema?: VbenFormSchema[];
  /** 表格列配置 */
  columns: any[];
  /** 数据加载函数 */
  request: (params: any) => Promise<{ list: T[]; total: number }>;
  /** 工具栏 */
  toolbar?: boolean;
  /** 操作栏 */
  actionColumn?: any;
}

const props = defineProps<TableProProps<T>>();

const emit = defineEmits<{
  'row-click': [row: T];
  'selection-change': [rows: T[]];
}>();

// 表格数据
const dataSource = ref<T[]>([]);
const loading = ref(false);
const total = ref(0);
const page = ref(1);
const pageSize = ref(10);

// 查询表单
const [SearchForm, searchFormApi] = useVbenForm({
  schema: props.searchSchema || [],
  layout: 'inline',
  actionWrapperClass: '',
  wrapperClass: 'grid-cols-1 md:grid-cols-2 lg:grid-cols-3',
});

// 加载数据
const fetchData = async (params?: any) => {
  loading.value = true;
  try {
    const searchParams = await searchFormApi.getValues();
    const result = await props.request({
      page: page.value,
      pageSize: pageSize.value,
      ...searchParams,
      ...params,
    });

    dataSource.value = result.list;
    total.value = result.total;
  } catch (error) {
    console.error('加载数据失败:', error);
  } finally {
    loading.value = false;
  }
};

// 查询
const handleSearch = async () => {
  page.value = 1;
  await fetchData();
};

// 重置
const handleReset = async () => {
  await searchFormApi.resetForm();
  page.value = 1;
  await fetchData();
};

// 页码变化
const handlePageChange = (newPage: number, newPageSize: number) => {
  page.value = newPage;
  pageSize.value = newPageSize;
  fetchData();
};

// 刷新
const refresh = () => {
  fetchData();
};

// 暴露方法
defineExpose({
  refresh,
  getDataSource: () => dataSource.value,
  getSelectedRows: () => [], // 根据需要实现
});

// 初始化
watch(
  () => props.request,
  () => {
    fetchData();
  },
  { immediate: true },
);
</script>

<template>
  <Page>
    <!-- 查询表单 -->
    <template v-if="searchSchema && searchSchema.length > 0" #header>
      <div class="bg-card rounded-lg p-4">
        <SearchForm>
          <template #default>
            <div class="flex justify-end gap-2">
              <VbenButton @click="handleReset"> 重置 </VbenButton>
              <VbenButton type="primary" @click="handleSearch">
                查询
              </VbenButton>
            </div>
          </template>
        </SearchForm>
      </div>
    </template>

    <!-- 工具栏 -->
    <div v-if="toolbar || $slots.toolbar" class="mb-4 flex justify-between">
      <div>
        <slot name="toolbar" />
      </div>

      <div class="flex gap-2">
        <VbenButton @click="refresh"> 刷新 </VbenButton>
        <slot name="toolbarExtra" />
      </div>
    </div>

    <!-- 数据表格 -->
    <VbenTable
      :columns="columns"
      :data-source="dataSource"
      :loading="loading"
      :pagination="{
        current: page,
        pageSize,
        total,
        showSizeChanger: true,
        showQuickJumper: true,
        onChange: handlePageChange,
      }"
      row-key="id"
      @row-click="(row) => emit('row-click', row)"
    >
      <!-- 透传所有插槽 -->
      <template v-for="(_, name) in $slots" #[name]="slotProps">
        <slot :name="name" v-bind="slotProps" />
      </template>
    </VbenTable>
  </Page>
</template>
```
