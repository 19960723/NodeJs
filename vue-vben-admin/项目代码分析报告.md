
#### 4.2.2 Composable 规范

```typescript
// use-my-feature.ts
import { computed, ref } from 'vue';
import type { Ref } from 'vue';

export interface UseMyFeatureOptions {
  initialValue?: string;
  onSuccess?: (value: string) => void;
}

export interface UseMyFeatureReturn {
  value: Ref<string>;
  loading: Ref<boolean>;
  fetch: () => Promise<void>;
}

export function useMyFeature(
  options: UseMyFeatureOptions = {},
): UseMyFeatureReturn {
  const { initialValue = '', onSuccess } = options;

  const value = ref(initialValue);
  const loading = ref(false);

  async function fetch() {
    loading.value = true;
    try {
      // 逻辑实现
      onSuccess?.(value.value);
    } finally {
      loading.value = false;
    }
  }

  return {
    value,
    loading,
    fetch,
  };
}
```

#### 4.2.3 API 类规范

```typescript
// my-api.ts
import { Store } from '@vben-core/shared/store';
import { bindMethods } from '@vben-core/shared/utils';

export interface MyApiState {
  visible: boolean;
  loading: boolean;
}

export class MyApi {
  public store: Store<MyApiState>;
  public state: MyApiState;

  constructor(options: Partial<MyApiState> = {}) {
    this.store = new Store<MyApiState>(
      {
        visible: false,
        loading: false,
        ...options,
      },
      {
        onUpdate: () => {
          this.state = this.store.state;
        },
      },
    );

    this.state = this.store.state;

    // 绑定 this
    bindMethods(this);
  }

  open() {
    this.store.setState({ visible: true });
  }

  close() {
    this.store.setState({ visible: false });
  }

  setState(state: Partial<MyApiState>) {
    this.store.setState(state);
  }
}
```

```typescript
import { Store } from '@vben-core/shared/store';
import { bindMethods } from '@vben-core/shared/utils';
import type { DataGridProps } from './types';

export class DataGridApi {
  public store: Store<DataGridProps>;
  public state: DataGridProps;

  constructor(options: DataGridProps) {
    this.store = new Store<DataGridProps>(
      { ...options },
      {
        onUpdate: () => {
          this.state = this.store.state;
        },
      },
    );

    this.state = this.store.state;
    bindMethods(this);
  }

  async reload() {
    if (!this.state.api) return;

    this.setLoading(true);
    try {
      const data = await this.state.api({
        page: this.state.pagination?.current,
        pageSize: this.state.pagination?.pageSize,
      });
      this.store.setState({ data });
    } finally {
      this.setLoading(false);
    }
  }

  setLoading(loading: boolean) {
    this.store.setState({ loading });
  }

  getData() {
    return this.state.data || [];
  }
}
```

### 5.5 创建 Composable

#### 示例: 分页 Hook

```typescript
// hooks/use-pagination.ts
import { computed, ref } from 'vue';
import type { Ref } from 'vue';

export interface UsePaginationOptions {
  page?: number;
  pageSize?: number;
  total?: number;
  onChange?: (page: number, pageSize: number) => void;
}

export function usePagination(options: UsePaginationOptions = {}) {
  const page = ref(options.page || 1);
  const pageSize = ref(options.pageSize || 10);
  const total = ref(options.total || 0);

  const totalPages = computed(() => {
    return Math.ceil(total.value / pageSize.value);
  });

  function setPage(newPage: number) {
    page.value = newPage;
    options.onChange?.(page.value, pageSize.value);
  }

  function setPageSize(newSize: number) {
    pageSize.value = newSize;
    page.value = 1;
    options.onChange?.(page.value, pageSize.value);
  }

  function reset() {
    page.value = 1;
  }

  return {
    page,
    pageSize,
    total,
    totalPages,
    setPage,
    setPageSize,
    reset,
  };
}

// 使用
const { page, pageSize, total, setPage } = usePagination({
  pageSize: 20,
  onChange: (page, pageSize) => {
    console.log('页码变化', page, pageSize);
  },
});
```

## 9. 总结

### 9.1 项目核心优势

1. **高度模块化**: 核心功能与业务逻辑完全分离
2. **类型安全**: 全量 TypeScript，严格类型检查
3. **多 UI 支持**: 灵活切换 UI 框架
4. **强大的表单系统**: 基于 Schema 配置，支持复杂验证和联动
5. **完善的布局系统**: 开箱即用的后台布局方案
6. **优秀的开发体验**: Monorepo + Vite + HMR

### 9.2 开发扩展组件建议

1. **遵循现有规范**: 参考现有组件的目录结构和代码风格
2. **类型优先**: 先定义类型，再实现逻辑
3. **API 设计**: 对于复杂组件，使用 API 类 + Composable 模式
4. **样式使用 TailwindCSS**: 尽量避免自定义 CSS
5. **文档和测试**: 为组件编写文档和测试用例
6. **示例驱动**: 在 playground 中创建使用示例

### 9.3 学习路径建议

1. **熟悉项目结构**: 了解 Monorepo 组织方式
2. **研究 playground**: 查看示例项目的实现
3. **阅读核心代码**: 学习 `@core` 包的设计思路
4. **实践扩展**: 从简单组件开始，逐步掌握 API 类模式
5. **贡献代码**: 参与社区贡献，提升技能
