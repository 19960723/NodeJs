## 🚀 快速开始

### 1. 创建简单组件（5 分钟）

```bash
# 在 packages/effects/common-ui/src/components/ 创建目录
mkdir my-card
cd my-card
```

**创建 3 个文件**：

**`index.ts`**

```typescript
export { default as MyCard } from './my-card.vue';
export * from './types';
```

**`types.ts`**

```typescript
export interface MyCardProps {
  title?: string;
  description?: string;
}
```

**`my-card.vue`**

```vue
<script setup lang="ts">
import type { MyCardProps } from './types';

interface Props extends MyCardProps {}

withDefaults(defineProps<Props>(), {
  title: '',
  description: '',
});
</script>

<template>
  <div class="bg-card rounded-lg border p-4 shadow-sm">
    <h3 class="text-lg font-semibold">{{ title }}</h3>
    <p class="text-muted-foreground">{{ description }}</p>
    <slot />
  </div>
</template>
```

**导出组件**（在 `components/index.ts`）：

```typescript
export * from './my-card';
```

**使用**：

```vue
<script setup>
import { MyCard } from '@vben/common-ui';
</script>

<template>
  <MyCard title="标题" description="描述">
    <p>内容</p>
  </MyCard>
</template>
```

### 表单 API

```typescript
// 设置单个值
formApi.setFieldValue('username', 'admin');

// 设置多个值
formApi.setValues({ username: 'admin', role: 'admin' });

// 获取值
const value = formApi.getFieldValue('username');
const values = formApi.getValues();

// 验证
await formApi.validate();

// 重置
formApi.resetForm();

// 提交
await formApi.submitForm();
```

### 动态表单（字段联动）

```typescript
const [Form] = useVbenForm({
  schema: [
    {
      component: 'Select',
      fieldName: 'country',
      label: '国家',
      componentProps: {
        options: [
          { label: '中国', value: 'china' },
          { label: '美国', value: 'usa' },
        ],
      },
    },
    {
      component: 'Select',
      fieldName: 'city',
      label: '城市',
      // 👇 依赖 country 字段
      dependencies: {
        triggerFields: ['country'],
        // 根据 country 动态改变选项
        componentProps(values) {
          const cityOptions = {
            china: [
              { label: '北京', value: 'beijing' },
              { label: '上海', value: 'shanghai' },
            ],
            usa: [
              { label: '纽约', value: 'new-york' },
              { label: '洛杉矶', value: 'los-angeles' },
            ],
          };

          return {
            options: cityOptions[values.country] || [],
          };
        },
        // 控制显示/隐藏
        if(values) {
          return !!values.country;
        },
        show: true,
      },
    },
  ],
});
```

### 自定义验证（Zod）

```typescript
import { z } from 'zod';

const [Form] = useVbenForm({
  schema: [
    {
      component: 'InputPassword',
      fieldName: 'password',
      label: '密码',
      rules: z
        .string()
        .min(8, '密码至少8位')
        .regex(/[A-Z]/, '必须包含大写字母')
        .regex(/[0-9]/, '必须包含数字'),
    },
    {
      component: 'InputPassword',
      fieldName: 'confirmPassword',
      label: '确认密码',
      dependencies: {
        triggerFields: ['password'],
        rules(values) {
          return z
            .string()
            .refine((val) => val === values.password, '两次密码不一致');
        },
      },
    },
  ],
});
```

## 🎨 弹窗组件速查

### Modal（模态框）

```typescript
import { useVbenModal } from '@vben-core/popup-ui';

const [Modal, modalApi] = useVbenModal({
  title: '标题',
  onConfirm: async () => {
    // 确认逻辑
    console.log('确认');
  },
  onCancel: () => {
    console.log('取消');
  },
});

// 打开/关闭
modalApi.open();
modalApi.close();

// 动态修改标题
modalApi.setState({ title: '新标题' });
```

**带表单的 Modal**：

```vue
<script setup lang="ts">
import { useVbenModal } from '@vben-core/popup-ui';
import { useVbenForm } from '#/adapter/form';

const [Form, formApi] = useVbenForm({
  schema: [
    { component: 'Input', fieldName: 'name', label: '姓名', rules: 'required' },
  ],
});

const [Modal, modalApi] = useVbenModal({
  title: '添加用户',
  onConfirm: async () => {
    await formApi.validate();
    const values = formApi.getValues();
    console.log(values);
    modalApi.close();
  },
});

function handleOpen() {
  modalApi.open();
}
</script>

<template>
  <button @click="handleOpen">打开</button>
  <Modal>
    <Form />
  </Modal>
</template>
```

## 📦 组件注册（表单适配器）

### 注册自定义组件到表单

**步骤 1**: 创建组件

```vue
<!-- components/my-input.vue -->
<script setup lang="ts">
const modelValue = defineModel<string>();
</script>

<template>
  <input v-model="modelValue" class="flex h-10 w-full rounded-md border" />
</template>
```

**步骤 2**: 在 adapter 中注册

```typescript
// playground/src/adapter/form.ts
import { setupVbenForm, useVbenForm as useForm } from '@vben-core/form-ui';
import MyInput from './components/my-input.vue';

setupVbenForm({
  config: {
    baseModelPropName: 'value',
    modelPropNameMap: {
      MyInput: 'modelValue', // 👈 指定 v-model 属性名
    },
  },
  components: {
    MyInput, // 👈 注册组件
  },
});

export { useForm as useVbenForm };

// 👇 类型增强
declare module '@vben-core/form-ui' {
  export interface VbenFormSchema {
    component?: 'Input' | 'Select' | 'MyInput'; // 👈 添加类型
  }
}
```

**步骤 3**: 使用

```typescript
const [Form] = useVbenForm({
  schema: [
    {
      component: 'MyInput', // 👈 现在可以使用了
      fieldName: 'myField',
      label: '自定义输入框',
    },
  ],
});
```

## 🚦 常见场景解决方案

### 场景 1: 创建 CRUD 页面

```vue
<script setup lang="ts">
import { ref } from 'vue';
import { Page } from '@vben/common-ui';
import { useVbenForm } from '#/adapter/form';
import { useVbenModal } from '@vben-core/popup-ui';

// 表格数据
const data = ref([]);

// 表单
const [Form, formApi] = useVbenForm({
  schema: [
    { component: 'Input', fieldName: 'name', label: '名称', rules: 'required' },
  ],
});

// 弹窗
const [Modal, modalApi] = useVbenModal({
  title: '添加',
  onConfirm: async () => {
    await formApi.validate();
    const values = formApi.getValues();
    // 保存数据
    console.log(values);
    modalApi.close();
  },
});

function handleAdd() {
  formApi.resetForm();
  modalApi.open();
}

function handleEdit(record: any) {
  formApi.setValues(record);
  modalApi.setState({ title: '编辑' });
  modalApi.open();
}
</script>

<template>
  <Page title="列表页">
    <template #extra>
      <button @click="handleAdd">添加</button>
    </template>

    <!-- 表格 -->
    <table>
      <tr v-for="item in data" :key="item.id">
        <td>{{ item.name }}</td>
        <td>
          <button @click="handleEdit(item)">编辑</button>
        </td>
      </tr>
    </table>

    <!-- 弹窗 -->
    <Modal>
      <Form />
    </Modal>
  </Page>
</template>
```

## 🎉 总结

### 开发流程

1. **创建组件** → 在 `packages/effects/common-ui/src/components/` 下创建
2. **导出组件** → 在 `index.ts` 中导出
3. **注册组件** → （可选）在 adapter 中注册到表单
4. **使用组件** → 在页面中导入使用
5. **添加示例** → 在 playground 中添加示例

### 学习建议

1. 先看 playground 示例项目
2. 参考现有组件的实现
3. 从简单组件开始实践
4. 逐步掌握 API 类模式
5. 阅读官方文档
